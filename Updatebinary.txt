UPDATE BINARY FILE

AIM:
Create a binary file studentdts.dat with rollno, name and marks. Input a rollno and update the marks.






PROCEDURE:



import pickle

def creating():
    f = open("e:\\studentdts.dat", 'ab')
    record = []
    while True:
        roll = int(input("Enter student roll no: "))
        name = input("Enter student Name: ")
        marks = float(input("Enter Marks: "))
        data = [roll, name, marks]
        record.append(data)
        choice = input("Wish to add more records? ")
        if choice.upper() == 'N':
            break
    pickle.dump(record, f)
    print("Record Added")
    f.close()

def updating(rno):
    f = open("e:\\studentdts.dat", 'rb+')
    record = pickle.load(f)
    found = False
    for rec in record:
        if rec[0] == rno:
            print("Current Mark is: ", rec[2])
            rec[2] = float(input("New Mark: "))
            found = True
            break
            
    if found:
        f.seek(0)
        pickle.dump(record, f)
        print("Mark Updated")
    f.close()

def display():
    f = open("e:\\studentdts.dat", 'rb+')
    record = pickle.load(f)
    for rec in record:
        print(rec)
    f.close()

# Main Program
ch = 'y'
while ch == 'y':
    print("MENU")
    print("1: create")
    print("2: update")
    print("3: display")
    choice = int(input("Enter your choice: "))
    if choice == 1:
        creating()
    elif choice == 2:
        rno = int(input("Enter roll number to search: "))
        updating(rno)
    elif choice == 3:
        display()
    else:
        print("Invalid choice")
    ch = input("Do you want to continue or not: ")
